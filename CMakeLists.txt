cmake_minimum_required(VERSION 3.14)
project(chroma_apex VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(chroma_apex SHARED core/src/chroma_apex.c)
set_target_properties(chroma_apex PROPERTIES VERSION 0.1.0 SOVERSION 0)
target_compile_definitions(chroma_apex PRIVATE CHROMA_APEX_BUILD)
target_include_directories(chroma_apex PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/core/include> $<INSTALL_INTERFACE:include>)
if(NOT MSVC)
  target_compile_options(chroma_apex PRIVATE -O3 -Wall -Wextra -Wpedantic)
endif()
add_executable(apex_test tests/test_c.c)
target_link_libraries(apex_test PRIVATE chroma_apex)
if(UNIX)
  target_link_libraries(apex_test PRIVATE m)
endif()
enable_testing()
add_test(NAME apex_test COMMAND apex_test)
add_executable(apex_benchmark bench/benchmark.cpp)
target_link_libraries(apex_benchmark PRIVATE chroma_apex)
target_compile_features(apex_benchmark PRIVATE cxx_std_17)
install(TARGETS chroma_apex LIBRARY DESTINATION lib RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
install(FILES core/include/chroma_apex.h DESTINATION include)

option(APX_WITH_IPS4O "Benchmark separately supplied upstream IPS4o header" OFF)
if(APX_WITH_IPS4O)
  if(NOT APX_IPS4O_INCLUDE_DIR)
    message(FATAL_ERROR "Set -DAPX_IPS4O_INCLUDE_DIR=/path/to/upstream/ips4o")
  endif()
  target_include_directories(apex_benchmark PRIVATE "${APX_IPS4O_INCLUDE_DIR}")
  target_compile_definitions(apex_benchmark PRIVATE APX_USE_IPS4O)
endif()
