cmake_minimum_required(VERSION 3.14)
project(chroma_apex_llm LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(apex_llm SHARED src/apex_llm.cpp)
target_include_directories(apex_llm PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(apex_llm PRIVATE /O2 /W4)
else()
  target_compile_options(apex_llm PRIVATE -O3 -Wall -Wextra -Wpedantic)
endif()
add_executable(apex_llm_test tests/test_sampling.cpp)
target_link_libraries(apex_llm_test PRIVATE apex_llm)
add_executable(apex_llm_benchmark bench/bench_sampling.cpp)
target_link_libraries(apex_llm_benchmark PRIVATE apex_llm)
enable_testing()
add_test(NAME sampler_correctness COMMAND apex_llm_test)

option(APX_WITH_LLAMA_CPP "Build optional llama.cpp adapter (requires installed public llama.h and libllama)" OFF)
if(APX_WITH_LLAMA_CPP)
  find_path(LLAMA_INCLUDE_DIR llama.h REQUIRED)
  find_library(LLAMA_LIBRARY llama REQUIRED)
  add_library(apex_llama_cpp_adapter SHARED adapters/llama_cpp_sampler.cpp)
  target_include_directories(apex_llama_cpp_adapter PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR}/adapters ${LLAMA_INCLUDE_DIR})
  target_link_libraries(apex_llama_cpp_adapter PRIVATE apex_llm ${LLAMA_LIBRARY})
endif()
